\chapter{L4 — Instances of the computability structure}

% P-R reference: Chapter 2, examples 2.1, 2.4-2.5, literature/papers/PourEl-Richards-chapt2.md.

\begin{theorem}
  \label{thm:l4_cmap_instance}
  \lean{ComputableAnalysis.L4.computabilityStructureCMap_of}
  \uses{def:l3_computabilityStructure}
  $C([a,b], \mathbb{R})$ with the sup-norm carries a \texttt{ComputabilityStructure}~$\mathbb{R}$, where a sequence $(s_n)$ in $C[a,b]$ is computable iff there is a computable double array of rational-coefficient polynomials $p_{n,m}$ with $\|s_n - p_{n,m}\|_\infty \le 2^{-m}$ for all $n,m$. The construction is parameterized by an explicit rational bound $B \ge \max(|a|,|b|)$, matching P-R Ch.\,2:128's restriction to recursive endpoints. P-R Ch.\,2:141--153.
\end{theorem}
\begin{proof}
  \uses{def:l1_isComputableSeqReal}
  The three structure fields are verified separately.
  \emph{Linearity} (P-R Axiom~1, Linear Forms): given computable polynomial
  approximants for $(x_n)$ and $(y_n)$ and computable rational scalar arrays
  $\alpha, \beta$, the finite combination $\sum_{k\le d(n)} \alpha_{n,k} x_k +
  \beta_{n,k} y_k$ is approximated by the same combination of the approximating
  polynomials. A precision pad $M(n,m)$ absorbing the per-summand coefficient and
  sup-norm bounds turns the triangle inequality into the required $2^{-m}$
  estimate.
  \emph{Effective limit} (P-R Axiom~2): the diagonal of a computably-Cauchy
  double array of polynomial approximants is again a polynomial approximant of
  the limit (P-R Ch.\,0 Thm.\,4).
  \emph{Computable norm} (P-R Axiom~3): $\|s_n\|_\infty = \lim_m \|p_{n,m}\|_\infty$
  and the sup-norm of a rational polynomial on $[a,b]$ is a computable real, so
  the norms form a computable sequence of reals (P-R Ch.\,0 Thm.\,7).
\end{proof}
