\begin{definition}
  \label{def:l3_computabilityStructure}
  \notready
  Typeclass \texttt{ComputabilityStructure}~$\Bbbk$~$E$ on a Banach space $[NormedAddCommGroup\,E]$ $[NormedSpace\,\Bbbk\,E]$ $[CompleteSpace\,E]$, with three axioms:
  \begin{itemize}
    \item \textbf{A1 (linearity).} Computable sequences are closed under rational linear combinations.
    \item \textbf{A2 (effective limit).} An effective Cauchy sequence of computable sequences has a computable limit.
    \item \textbf{A3 (computable norm).} The norm of a computable sequence is a computable sequence of reals.
  \end{itemize}
  P-R Ch.\,2:18--22.
\end{definition}
