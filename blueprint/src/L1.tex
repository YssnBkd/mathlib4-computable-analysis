\begin{definition}
  \label{def:l1_isComputableSeqRat}
  \lean{ComputableAnalysis.L1.IsComputableSeqRat}
  \leanok
  \uses{def:isRecursive}
  Predicate \texttt{IsComputableSeqRat}\,$: (\mathbb{N} \to \mathbb{Q}) \to \texttt{Prop}$. A rational sequence $r$ is computable iff there exist recursive functions $a, b, s : \mathbb{N} \to \mathbb{N}$ with $b(k) \ne 0$ such that $r_k = (-1)^{s(k)} \cdot (a(k) / b(k))$. P-R Ch.\,0:46, Definition 1.
  See \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean:82}.
\end{definition}

\begin{definition}
  \label{def:l1_isComputableDoubleSeqRat}
  \lean{ComputableAnalysis.L1.IsComputableDoubleSeqRat}
  \leanok
  \uses{def:l1_isComputableSeqRat}
  Predicate \texttt{IsComputableDoubleSeqRat}\,$: (\mathbb{N} \times \mathbb{N} \to \mathbb{Q}) \to \texttt{Prop}$. A double sequence $r_{n,k}$ is computable iff its re-indexing through Mathlib's \texttt{Nat.unpair} is a computable rational sequence. P-R Ch.\,0:189 (``a double sequence will be called computable if it is mapped onto a computable sequence by one of the standard recursive pairing functions'').
  See \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean:92}.
\end{definition}

\begin{lemma}
  \label{lem:l1_isComputableSeqRat_const}
  \lean{ComputableAnalysis.L1.isComputableSeqRat_const}
  \leanok
  \uses{def:l1_isComputableSeqRat}
  For every rational $q$, the constant sequence $\lambda n.\,q$ satisfies \texttt{IsComputableSeqRat}. Witness $(a,b,s) := (|q.\mathrm{num}|, q.\mathrm{den}, \mathbf{1}[q.\mathrm{num} < 0])$, all constant hence recursive; the arithmetic identity reduces to a sign case-split on \texttt{Rat.num\_div\_den}.
\end{lemma}
\begin{proof}\leanok
  Formalized in \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean}.
\end{proof}

\begin{definition}
  \label{def:l1_isComputableSeqReal}
  \lean{ComputableAnalysis.L1.IsComputableSeqReal}
  \leanok
  \uses{def:l1_isComputableDoubleSeqRat}
  Predicate \texttt{IsComputableSeqReal}\,$: (\mathbb{N} \to \mathbb{R}) \to \texttt{Prop}$ on Mathlib's \texttt{Real}. A real sequence $x$ is computable iff there exists a computable rational double-sequence $r_{n,k}$ with $|r_{n,k} - x_n| \le 2^{-k}$ for all $n, k$. P-R Ch.\,0:203, Definition 5a.
  See \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean:101}.
\end{definition}

\begin{lemma}
  \label{lem:l1_isComputableSeqReal_const_rat}
  \lean{ComputableAnalysis.L1.isComputableSeqReal_const_rat}
  \leanok
  \uses{def:l1_isComputableSeqReal, lem:l1_isComputableSeqRat_const}
  For every rational $q$, the real-valued constant sequence $\lambda n.\,(q : \mathbb{R})$ satisfies \texttt{IsComputableSeqReal}. Witness: the constant double sequence $r(n,k) := q$. The error bound $|q - q| = 0 \le 2^{-k}$ is immediate.
\end{lemma}
\begin{proof}\leanok
  Formalized in \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean}.
\end{proof}

\begin{lemma}
  \label{lem:l1_isComputableSeqRat_add}
  \lean{ComputableAnalysis.L1.IsComputableSeqRat.add}
  \leanok
  \uses{def:l1_isComputableSeqRat}
  Closure of \texttt{IsComputableSeqRat} under pointwise addition: if $h_i : \texttt{IsComputableSeqRat}\,r_i$ for $i \in \{1,2\}$, then $r_1 + r_2$ (Pi-pointwise sum) is computable. The new $\mathbb{N}$-level witness $(\mathrm{newA}, \mathrm{newB}, \mathrm{newS})$ dispatches on the sign-parities $(s_1\,k \bmod 2, s_2\,k \bmod 2)$ and the cross-product comparison $a_1\,k \cdot b_2\,k \ge a_2\,k \cdot b_1\,k$; $\mathbb{N}$-truncated subtraction handles the sign-cancellation cases.
\end{lemma}
\begin{proof}\leanok
  Formalized in \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean}.
\end{proof}

\begin{lemma}
  \label{lem:l1_isComputableSeqRat_mul}
  \lean{ComputableAnalysis.L1.IsComputableSeqRat.mul}
  \leanok
  \uses{def:l1_isComputableSeqRat}
  Closure of \texttt{IsComputableSeqRat} under pointwise multiplication: $r_1 \cdot r_2$ is computable when both factors are. The witness is direct, $(\mathrm{newA}, \mathrm{newB}, \mathrm{newS}) := (a_1 \cdot a_2,\, b_1 \cdot b_2,\, s_1 + s_2)$; the identity reduces to a single \texttt{pow\_add} + \texttt{field\_simp} step (no sign-case dispatch).
\end{lemma}
\begin{proof}\leanok
  Formalized in \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean}.
\end{proof}

\begin{lemma}
  \label{lem:l1_isComputableSeqRat_comp}
  \lean{ComputableAnalysis.L1.IsComputableSeqRat.comp}
  \leanok
  \uses{def:l1_isComputableSeqRat}
  Closure of \texttt{IsComputableSeqRat} under precomposition with a computable re-indexing: if $\sigma : \mathbb{N} \to \mathbb{N}$ is \texttt{Computable} and $r : \mathbb{N} \to \mathbb{Q}$ satisfies \texttt{IsComputableSeqRat}, then so does $r \circ \sigma$. Each component $(a, b, s)$ of the witness is precomposed with $\sigma$; closure of \texttt{Computable} under composition (\texttt{Computable.comp}) supplies the recursion-theoretic data.
\end{lemma}
\begin{proof}\leanok
  Formalized in \texttt{ComputableAnalysis/L1/ComputableSeqReal.lean}.
\end{proof}
