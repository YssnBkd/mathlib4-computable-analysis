\begin{lemma}
  \label{lem:l4_cmap_axiom1}
  \lean{ComputableAnalysis.L4.isComputableSeqCMap_linearCombination}
  \leanok
  \uses{def:l3_computabilityStructure, def:l1_isComputableSeqRat}
  Fix real endpoints $\alpha, \beta \in \mathbb{R}$ and a rational bound $B \in \mathbb{N}$ with $|\alpha| \le B$ and $|\beta| \le B$. The C[a,b]-sequence predicate \texttt{IsComputableSeqCMap} on $C([\alpha,\beta], \mathbb{R})$ is closed under finite linear combinations with computable rational scalar arrays: given $(x_n), (y_n)$ both \texttt{IsComputableSeqCMap}, computable scalar double-arrays $\alpha', \beta' : \mathbb{N} \times \mathbb{N} \to \mathbb{R}$ (in the \texttt{ScalarComputableSeq}~$\mathbb{R}$ sense, after \texttt{Nat.unpair} re-indexing), and a computable arity $d : \mathbb{N} \to \mathbb{N}$, the sequence
  \[ s_n := \sum_{k = 0}^{d(n)} \bigl( \alpha'(n, k) \cdot x_k + \beta'(n, k) \cdot y_k \bigr) \]
  satisfies \texttt{IsComputableSeqCMap}. This is the C[a,b] specialisation of P-R Axiom~1 (Linear Forms), P-R Ch.\,2:66--72.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:l1_isComputableSeqRat_add, lem:l1_isComputableSeqRat_mul, lem:l1_isComputableSeqRat_comp}
  The witness construction is direct. Let $(a^X, d^X)$ and $(a^Y, d^Y)$ be the polynomial-approximant witnesses for $(x_n), (y_n)$, with $\|x_n - p^X_{n,m}\|_\infty \le 2^{-m}$ etc., where $p^X_{n,m}$ denotes the polynomial $\sum_{j \le d^X(n,m)} a^X(n,m,j) \cdot t^j$ on $[\alpha,\beta]$. Define the combined polynomial
  \[ p^s_{n,m} := \sum_{k=0}^{d(n)} \bigl( \alpha'(n,k) \cdot p^X_{k, M(n,m)} + \beta'(n,k) \cdot p^Y_{k, M(n,m)} \bigr) \]
  with precision pad $M(n,m) := m + (d(n) + 1) + \mathrm{bound\_max}(n) + 2$, where $\mathrm{bound\_max}(n)$ is a uniform $\mathbb{N}$-upper-bound combining the per-$k$ norms and coefficient magnitudes for both legs of the linear combination (computed in Lean as the sum $\mathrm{bound\_x\_k} + \mathrm{bound\_y\_k} + \mathrm{bound\_\alpha R\_at\_0} + \mathrm{bound\_\beta R\_at\_0} + 6$, recovered from the inner $a$-witnesses of the L1 \texttt{IsComputableSeqRat} data).

  \emph{Computability of the witness array.} The rational double-sequence underlying $p^s_{n,m}$ is built from $a^X, a^Y, \alpha', \beta'$ via finite sums and products, after the polynomial multiplication $\alpha'(n,k) \cdot \sum_j a^X(k,M,j) t^j$ expands to a degree-by-degree convolution. Closure of \texttt{IsComputableSeqRat} under pointwise addition (\Cref{lem:l1_isComputableSeqRat_add}), multiplication (\Cref{lem:l1_isComputableSeqRat_mul}), and precomposition with a computable re-indexing (\Cref{lem:l1_isComputableSeqRat_comp}) supplies the recursion-theoretic witness. The bound $d^s_{n,m} := d(n) + \max_{k \le d(n)}(d^X(k,M) + d^Y(k,M))$ is computable by hypothesis on $d, d^X, d^Y$.

  \emph{Norm bound.} By the triangle inequality and homogeneity of the sup-norm, each summand $|\alpha'(n,k)| \cdot \|x_k - p^X_{k,M(n,m)}\|_\infty + |\beta'(n,k)| \cdot \|y_k - p^Y_{k,M(n,m)}\|_\infty$ is bounded by $\mathrm{bound\_max}(n) \cdot 2^{-M(n,m)}$ (using the per-$k$ closure of $\mathrm{bound\_max}(n)$ over both legs), hence
  \[ \|s_n - p^s_{n,m}\|_\infty \le \sum_{k=0}^{d(n)} \mathrm{bound\_max}(n) \cdot 2^{-M(n,m)} = (d(n) + 1) \cdot \mathrm{bound\_max}(n) \cdot 2^{-M(n,m)}. \]
  The pad's definition of $M(n,m)$ is chosen so that $(d(n)+1) \cdot \mathrm{bound\_max}(n) \cdot 2^{m - M(n,m)} \le 1$, yielding $\|s_n - p^s_{n,m}\|_\infty \le 2^{-m}$. The Lean proof's \texttt{closeout\_nat} lemma (\texttt{CMap.lean:477}) discharges this final arithmetic inequality.
\end{proof}

\begin{theorem}
  \label{thm:l4_cmap_instance}
  \lean{ComputableAnalysis.L4.computabilityStructureCMap_of}
  \leanok
  \uses{def:l3_computabilityStructure}
  $C([a,b], \mathbb{R})$ with the sup-norm carries a \texttt{ComputabilityStructure}~$\mathbb{R}$, where a sequence $(s_n)$ in $C[a,b]$ is computable iff there is a computable double array of rational-coefficient polynomials $p_{n,m}$ with $\|s_n - p_{n,m}\|_\infty \le 2^{-m}$ for all $n,m$. The construction is parameterized by an explicit rational bound $B \ge \max(|a|,|b|)$, computable-real witnesses for $\alpha, \beta$, and $\alpha \le \beta$, matching P-R Ch.\,2:128's restriction to recursive endpoints. P-R Ch.\,2:141--153.
\end{theorem}
\begin{proof}\leanok
  \uses{lem:l4_cmap_axiom1, lem:l4_cmap_axiom2, lem:l4_cmap_axiom3, def:l1_isComputableSeqReal}
  The three structure fields are verified separately.
  \emph{Linearity} (P-R Axiom~1, Linear Forms): by \Cref{lem:l4_cmap_axiom1}.
  \emph{Effective limit} (P-R Axiom~2): by \Cref{lem:l4_cmap_axiom2}.
  \emph{Computable norm} (P-R Axiom~3): delegates to \Cref{lem:l4_cmap_axiom3}, instantiated with the parameters $B, h_{\alpha\le B}, h_{\beta\le B}, h_{\alpha\text{-c}}, h_{\beta\text{-c}}, h_{\alpha\beta}$ supplied to the structure.
\end{proof}

\begin{lemma}
  \label{lem:l4_cmap_axiom2}
  \lean{ComputableAnalysis.L4.isComputableSeqCMap_of_effectiveLimit}
  \leanok
  \uses{def:l3_computabilityStructure, def:l1_isComputableSeqRat}
  C[a,b] computability is closed under effective uniform limits (P-R Ch.\,0 Theorem~4, \texttt{literature/papers/PourEl-Richards-chapt0.md:748}): if $\{f_{n,k}\}$ is an \texttt{IsComputableSeqCMap} double sequence and a computable modulus $e : \mathbb{N} \times \mathbb{N} \to \mathbb{N}$ witnesses $\|f_{n,k} - f_n\|_\infty \le 2^{-N}$ for $k \ge e(n,N)$, then $\{f_n\}$ is \texttt{IsComputableSeqCMap}.
\end{lemma}
\begin{proof}
  \leanok
  Diagonalise the polynomial-approximant double sequence by $a'_{n,N,j} := a_{n, e(n,N+1), N+1, j}$ and $d'_{n,N} := d_{n, e(n,N+1), N+1}$ (where $a, d$ are the rational-polynomial witnesses of the input double sequence). Computability of $(a', d')$ follows from \texttt{FinsetSumHelper.isComputableSeqRat\_tripleApply} (for $a'$) and the composability of \texttt{Computable} (for $d'$). The norm bound is the triangle estimate $\|f_n - p'_{n,N}\|_\infty \le \|f_n - f_{n,e(n,N+1)}\|_\infty + \|f_{n,e(n,N+1)} - p^{(n,k)}_{N+1}\|_\infty \le 2^{-(N+1)} + 2^{-(N+1)} = 2^{-N}$.
\end{proof}

\begin{lemma}
  \label{lem:l4_cmap_axiom3}
  \lean{ComputableAnalysis.L4.isComputableSeqCMap_norm}
  \leanok
  \uses{def:l3_computabilityStructure, def:l1_isComputableSeqReal, lem:l1_isComputableSeqReal_of_effectiveConvergence, lem:l1_isComputableSeqRat_max, lem:l1_isComputableSeqRat_abs}
  Sup-norm of a \texttt{IsComputableSeqCMap} family is a \texttt{IsComputableSeqReal} (P-R Ch.\,0 Theorem~7, Maximum Values, \texttt{literature/papers/PourEl-Richards-chapt0.md:977}). The signature takes a rational bound $B$ with $|\alpha|, |\beta| \le B$, computable-real witnesses $\texttt{IsComputableReal}~\alpha, \beta$ (for the grid construction), and the well-orderedness assumption $\alpha \le \beta$ (required to rule out a degenerate empty-interval adversary).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:l1_isComputableSeqReal_of_effectiveConvergence}
  Three-error decomposition for $|s_{n,k} - \|f_n\||_\infty \le 2^{-N}$ when $k \ge N$, following P-R chapt0:984--1012.
  \begin{itemize}
    \item Set $K := k + 2$ and let $p := \texttt{polyApproxCMap}~a~d~n~K$ be the $K$-th polynomial approximant of $f_n$.
    \item (a) \emph{Polynomial-approximation error}: $|\|p\| - \|f_n\|| \le \|p - f_n\| \le 2^{-K}$, by the input \texttt{herr} bound at precision $K$.
    \item (b) \emph{Sample/norm gap} $|s_{n,k} - \|p\|| \le 2 \cdot 2^{-K}$, decomposed:
      \begin{itemize}
        \item \emph{$h_{b1}$ (clamp)}: each grid sample $|p_\mathbb{R}(xQ_J)|$ is bounded by $\|p\| + 2^{-K}$ via the clamp $xc_J := \mathtt{projIcc}~\alpha~\beta~h_{\alpha\beta}(xQ_J) \in [\alpha,\beta]$, Lipschitz bound $|p_\mathbb{R}(xQ_J) - p(xc_J)| \le \mathrm{Lip} \cdot 2^{-mm} \le 2^{-K}$, and $|p(xc_J)| \le \|p\|$.
        \item \emph{$h_{b2}$ (cover)}: for each $x \in [\alpha,\beta]$, find $J^* \le kg$ with $|x - xQ_{J^*}|$ controlled. Non-degenerate $\alpha' < \beta'$: clamp $x$ into $[\alpha',\beta']$, parameterize $t \in [0,1]$, set $J^* := \lfloor kg \cdot t \rfloor$, giving $|x - xQ_{J^*}| \le 2^{-mm} + 2(B+1)/kg$. Degenerate $\alpha' \ge \beta'$ (forces $\beta - \alpha \le 2 \cdot 2^{-mm}$): midpoint case-split on $x \le (\alpha+\beta)/2$, pick $J^* := 0$ or $J^* := kg$, giving $|x - xQ_{J^*}| \le 2 \cdot 2^{-mm}$.
      \end{itemize}
    \item Summed errors: $|s_{n,k} - \|f_n\|| \le 2^{-K} + 2 \cdot 2^{-K} = 3 \cdot 2^{-(k+2)} \le 2^{-k} \le 2^{-N}$ when $k \ge N$.
  \end{itemize}
  Effective convergence then yields $\|f_n\|$ as \texttt{IsComputableSeqReal} via \Cref{lem:l1_isComputableSeqReal_of_effectiveConvergence}.
\end{proof}
